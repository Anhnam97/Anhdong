function combineLine_and_save(variable) {
    // Xử lý dồn về 1 hàng, chỉ gọi khi text có nhiều hàng
    variable = variable.replace(/[\n\t]+/g, ' ') // Thay thế các ký tự xuống dòng và tab bằng khoảng trắng
                       .replace(/\s+/g, ' ') // Thay thế nhiều khoảng trắng thành một khoảng trắng
                       .trim(); // Loại bỏ khoảng trắng ở đầu và cuối
    return variable; // Trả về kết quả
}

function loadGlobalConfig() {
    // Tải tốc độ từ localStorage
    try {
        const savedSpeed = localStorage.getItem('auto_speed');
        if (savedSpeed) {
            autoSpeed = parseFloat(savedSpeed);
            delayToNextFriend = autoSpeed * 1000;
            console.log('✅ Đã tải tốc độ từ localStorage:', autoSpeed + 's');
        }
    } catch(e) {}
    
    // Tải giới tính từ localStorage
    try {
        const savedGender = localStorage.getItem('gender_filter');
        if (savedGender) {
            selectedGender = savedGender;
            genderFilterEnabled = savedGender !== 'ALL';
            console.log('✅ Đã tải lọc giới tính:', selectedGender);
        }
    } catch(e) {}
}

// === GỌI HÀM TẢI CẤU HÌNH ===
loadGlobalConfig();

function updateAutoSpeed(newSpeed) {
    if (newSpeed >= autoSpeedMin && newSpeed <= autoSpeedMax) {
        autoSpeed = newSpeed;
        delayToNextFriend = autoSpeed * 1000;
        localStorage.setItem('auto_speed', autoSpeed);
        saveSingleLocalData('auto_speed', autoSpeed);
        console.log('✅ Đã cập nhật tốc độ Auto:', autoSpeed + 's');
        return true;
    }
    return false;
}

// ============================================
// === THÊM HÀM KIỂM TRA VÀ CẬP NHẬT GIỚI TÍNH ===
// ============================================
function updateGenderFilter(newGender) {
    const validGenders = ['ALL', 'MALE', 'FEMALE'];
    if (validGenders.includes(newGender)) {
        selectedGender = newGender;
        genderFilterEnabled = newGender !== 'ALL';
        localStorage.setItem('gender_filter', selectedGender);
        saveSingleLocalData('gender_filter', selectedGender);
        console.log('✅ Đã cập nhật lọc giới tính:', selectedGender);
        return true;
    }
    return false;
}
